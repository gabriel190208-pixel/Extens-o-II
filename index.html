import java.time.LocalDateTime; import java.time.format.DateTimeFormatter; import java.util.ArrayList; import java.util.List; import java.util.Scanner; public class trabalho { // Classe que representa um Registro de Autocontrole (RF01, RF02, RF03) static class RegistroControle { private int id; private String tipoControle; private String produtorOuSetor; private String detalhesMedicao; private String responsavel; // <> Identificar responsável pelo registro private String dataHora; // <> Registrar data e horário public RegistroControle(int id, String tipoControle, String produtorOuSetor, String detalhesMedicao, String responsavel) { this.id = id; this.tipoControle = tipoControle; this.produtorOuSetor = produtorOuSetor; this.detalhesMedicao = detalhesMedicao; this.responsavel = responsavel; // Gravação automática da data e horário atual no momento do registro DateTimeFormatter formatter = DateTimeFormatter.ofPattern("dd/MM/yyyy HH:mm:ss"); this.dataHora = LocalDateTime.now().format(formatter); } public int getId() { return id; } public String getTipoControle() { return tipoControle; } public String getProdutorOuSetor() { return produtorOuSetor; } public String getDetalhesMedicao() { return detalhesMedicao; } public String getResponsavel() { return responsavel; } public String getDataHora() { return dataHora; } public void setDetalhesMedicao(String detalhesMedicao) { this.detalhesMedicao = detalhesMedicao; } @Override public String toString() { return String.format("[ID: %d] %s | Data/Hora: %s | Produtor/Setor: %s | Dados: %s | Resp.: %s", id, tipoControle, dataHora, produtorOuSetor, detalhesMedicao, responsavel); } } private static List bancoRegistros = new ArrayList<>(); private static int proximoId = 1; private static Scanner scanner = new Scanner(System.in); public static void main(String[] args) { carregarDadosIniciais(); // Pré-carrega alguns registros para facilitar testes na apresentação int perfil = -1; while (perfil != 0) { System.out.println("\n======================================================="); System.out.println(" SISTEMA DE REGISTROS DE AUTOCONTROLE DO LATICÍNIO"); System.out.println("======================================================="); System.out.println("Selecione o perfil de acesso:"); System.out.println("1 - Funcionário do Laboratório (Operacional)"); System.out.println("2 - Responsável pelo Laboratório / Controle de Qualidade"); System.out.println("0 - Sair do Sistema"); System.out.print("Opção: "); try { perfil = Integer.parseInt(scanner.nextLine()); switch (perfil) { case 1: menuFuncionario(); break; case 2: menuResponsavelQualidade(); break; case 0: System.out.println("Encerrando o sistema..."); break; default: System.out.println("Opção inválida. Tente novamente."); } } catch (NumberFormatException e) { System.out.println("Erro: Digite apenas números."); } } scanner.close(); } // ========================================================================= // MÓDULO 1: REGISTROS (Ator: Funcionário do Laboratório) // ========================================================================= private static void menuFuncionario() { System.out.print("\nInforme o nome do Funcionário responsável pelo turno: "); String nomeFuncionario = scanner.nextLine().trim(); if (nomeFuncionario.isEmpty()) { nomeFuncionario = "Funcionário Padrão"; } int opcao = -1; while (opcao != 0) { System.out.println("\n--- MÓDULO DE REGISTROS DO LABORATÓRIO (Resp.: " + nomeFuncionario + ") ---"); System.out.println("1 - Registrar recebimento e análises do leite (RF01)"); System.out.println("2 - Registrar controles de produtos (RF02)"); System.out.println("3 - Registrar limpeza e higienização (RF02)"); System.out.println("4 - Registrar temperaturas (RF02)"); System.out.println("5 - Registrar expedição (RF02)"); System.out.println("6 - Registrar não conformidades (RF02)"); System.out.println("0 - Voltar ao menu principal"); System.out.print("Escolha o tipo de registro: "); try { opcao = Integer.parseInt(scanner.nextLine()); if (opcao >= 1 && opcao <= 6) { realizarRegistro(opcao, nomeFuncionario); } else if (opcao != 0) { System.out.println("Opção inválida."); } } catch (NumberFormatException e) { System.out.println("Erro: Digite apenas números."); } } } private static void realizarRegistro(int tipoOpcao, String responsavel) { String tipoControle = ""; String promptOrigem = "Informe o Setor / Equipamento: "; switch (tipoOpcao) { case 1: tipoControle = "Recebimento e Análises do Leite"; promptOrigem = "Informe o Produtor / Linha de Coleta: "; break; case 2: tipoControle = "Controle de Produtos"; promptOrigem = "Informe o Produto / Lote: "; break; case 3: tipoControle = "Limpeza e Higienização"; break; case 4: tipoControle = "Controle de Temperaturas"; promptOrigem = "Informe a Câmara Fria / Tanque: "; break; case 5: tipoControle = "Expedição"; promptOrigem = "Informe o Destino / Carga: "; break; case 6: tipoControle = "Não Conformidade"; break; } System.out.print(promptOrigem); String produtorOuSetor = scanner.nextLine(); System.out.print("Informe os dados/medições do controle (ex: Temp 3.8°C, Alizarol OK): "); String detalhes = scanner.nextLine(); // Cria o registro aplicando automaticamente o <> de responsável e data/hora (RF03) RegistroControle novo = new RegistroControle(proximoId++, tipoControle, produtorOuSetor, detalhes, responsavel); bancoRegistros.add(novo); System.out.println("\n[SUCESSO] Registro salvo com rastreabilidade automática:"); System.out.println(novo); } // ========================================================================= // MÓDULO 2 E 3: CONSULTAS, ACOMPANHAMENTO E RELATÓRIOS (Ator: Resp. Qualidade) // ========================================================================= private static void menuResponsavelQualidade() { int opcao = -1; while (opcao != 0) { System.out.println("\n--- MÓDULO DE CONTROLE DE QUALIDADE E SEGURANÇA ---"); System.out.println("1 - Consultar histórico digital completo (RF04)"); System.out.println("2 - Consultar com filtros por data, produtor ou tipo (RF05 - <>)"); System.out.println("3 - Acompanhar frequência dos controles (RF06)"); System.out.println("4 - Editar registro existente (RF04)"); System.out.println("5 - Gerar e imprimir relatório consolidado (RF07)"); System.out.println("6 - Realizar backup das informações (RNF01)"); System.out.println("0 - Voltar ao menu principal"); System.out.print("Opção: "); try { opcao = Integer.parseInt(scanner.nextLine()); switch (opcao) { case 1: consultarHistoricoCompleto(); break; case 2: consultarComFiltros(); break; case 3: acompanharFrequencia(); break; case 4: editarRegistro(); break; case 5: gerarRelatorio(); break; case 6: realizarBackup(); break; case 0: break; default: System.out.println("Opção inválida."); } } catch (NumberFormatException e) { System.out.println("Erro: Digite apenas números."); } } } private static void consultarHistoricoCompleto() { System.out.println("\n--- HISTÓRICO DIGITAL DE REGISTROS ---"); if (bancoRegistros.isEmpty()) { System.out.println("Nenhum registro encontrado."); return; } for (RegistroControle r : bancoRegistros) { System.out.println(r); } } private static void consultarComFiltros() { System.out.print("\nDigite o termo de filtro (Data dd/mm/aaaa, Produtor ou Tipo de Controle): "); String termo = scanner.nextLine().toLowerCase(); System.out.println("\n--- RESULTADOS FILTRADOS ---"); boolean encontrou = false; for (RegistroControle r : bancoRegistros) { if (r.getDataHora().toLowerCase().contains(termo) || r.getProdutorOuSetor().toLowerCase().contains(termo) || r.getTipoControle().toLowerCase().contains(termo)) { System.out.println(r); encontrou = true; } } if (!encontrou) { System.out.println("Nenhum registro corresponde ao filtro informado."); } } private static void acompanharFrequencia() { System.out.println("\n--- ACOMPANHAMENTO DE FREQUÊNCIA DOS CONTROLES ---"); System.out.println("Total de registros lançados no sistema: " + bancoRegistros.size()); long qtdLeite = bancoRegistros.stream().filter(r -> r.getTipoControle().contains("Leite")).count(); long qtdTemp = bancoRegistros.stream().filter(r -> r.getTipoControle().contains("Temperaturas")).count(); long qtdLimpeza = bancoRegistros.stream().filter(r -> r.getTipoControle().contains("Limpeza")).count(); System.out.println("- Recebimento e Análises do Leite: " + qtdLeite + " registro(s)"); System.out.println("- Controle de Temperaturas: " + qtdTemp + " registro(s)"); System.out.println("- Limpeza e Higienização: " + qtdLimpeza + " registro(s)"); System.out.println("Status: Verificação manual concluída pelo gestor."); } private static void editarRegistro() { consultarHistoricoCompleto(); System.out.print("\nInforme o ID do registro que deseja editar: "); try { int idBusca = Integer.parseInt(scanner.nextLine()); for (RegistroControle r : bancoRegistros) { if (r.getId() == idBusca) { System.out.println("Dados atuais: " + r.getDetalhesMedicao()); System.out.print("Informe os novos dados/medições corrigidos: "); String novosDados = scanner.nextLine(); r.setDetalhesMedicao(novosDados); System.out.println("[SUCESSO] Registro atualizado: " + r); return; } } System.out.println("Registro com ID " + idBusca + " não encontrado."); } catch (NumberFormatException e) { System.out.println("ID inválido."); } } private static void gerarRelatorio() { DateTimeFormatter formatter = DateTimeFormatter.ofPattern("dd/MM/yyyy HH:mm:ss"); System.out.println("\n======================================================="); System.out.println(" RELATÓRIO OFICIAL DE AUTOCONTROLE - LATICÍNIO"); System.out.println(" Emitido em: " + LocalDateTime.now().format(formatter)); System.out.println("======================================================="); consultarHistoricoCompleto(); System.out.println("======================================================="); System.out.println("[INFO] Documento formatado e enviado para fila de impressão."); } private static void realizarBackup() { DateTimeFormatter formatter = DateTimeFormatter.ofPattern("yyyyMMdd_HHmmss"); String nomeArquivo = "backup_autocontrole_" + LocalDateTime.now().format(formatter) + ".dat"; System.out.println("\n[BACKUP] Exportando " + bancoRegistros.size() + " registro(s)..."); System.out.println("[SUCESSO] Cópia de segurança gerada localmente: " + nomeArquivo); } // Dados iniciais para demonstração imediata private static void carregarDadosIniciais() { bancoRegistros.add(new RegistroControle(proximoId++, "Recebimento e Análises do Leite", "Produtor João Silva", "1.200L | Temp: 3.9°C | Acidez: 16°D | Alizarol: Negativo", "Carlos Mendes")); bancoRegistros.add(new RegistroControle(proximoId++, "Controle de Temperaturas", "Câmara Fria 01", "Temperatura aferida: 2.5°C (Conforme)", "Carlos Mendes")); bancoRegistros.add(new RegistroControle(proximoId++, "Limpeza e Higienização", "Tanque de Pasteurização", "CIP realizado com solução alcalina/ácida - OK", "Ana Paula")); } }
